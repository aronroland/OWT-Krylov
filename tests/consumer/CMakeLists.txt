cmake_minimum_required(VERSION 3.20)
project(OWTKrylovConsumer LANGUAGES CXX)

find_package(OWTKrylov CONFIG REQUIRED)
add_executable(vertex_consumer main.cpp)
target_link_libraries(vertex_consumer PRIVATE OWT::Krylov)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|IntelLLVM")
    foreach(mode IN ITEMS fast finite)
        add_executable(unsafe_math_${mode} EXCLUDE_FROM_ALL unsafe_math.cpp)
        target_link_libraries(unsafe_math_${mode} PRIVATE OWT::Krylov)
    endforeach()
    target_compile_options(unsafe_math_fast PRIVATE -ffast-math)
    target_compile_options(unsafe_math_finite PRIVATE -ffinite-math-only)
endif()
enable_testing()
add_test(NAME vertex_consumer_serial COMMAND vertex_consumer)
set_tests_properties(vertex_consumer_serial PROPERTIES TIMEOUT 120)
if(TARGET MPI::MPI_CXX)
    foreach(ranks IN ITEMS 2 4)
        add_test(NAME vertex_consumer_${ranks}_ranks
            COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks}
                $<TARGET_FILE:vertex_consumer>)
        set_tests_properties(vertex_consumer_${ranks}_ranks PROPERTIES TIMEOUT 120)
    endforeach()
endif()
